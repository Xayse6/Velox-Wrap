DROP TABLE IF EXISTS veiculos CASCADE;
DROP TABLE IF EXISTS modelos CASCADE;
DROP TABLE IF EXISTS marcas CASCADE;
DROP TABLE IF EXISTS usuarios CASCADE;

CREATE TABLE usuarios (
    id_usuario SERIAL PRIMARY KEY,
    nome_usuario VARCHAR(100) NOT NULL,
    cpf_usuario VARCHAR(14) NOT NULL UNIQUE,
    email_usuario VARCHAR(255) NOT NULL UNIQUE,
    senha_usuario VARCHAR(255) NOT NULL,
    tipo_usuario VARCHAR(20) NOT NULL DEFAULT 'usuario',
    CONSTRAINT tipo_usuario_check
        CHECK (tipo_usuario IN ('admin', 'usuario'))
);

CREATE TABLE marcas (
    id_marca SERIAL PRIMARY KEY,
    nome_marca VARCHAR(100) NOT NULL UNIQUE,
    sigla_marca CHAR(5) NOT NULL UNIQUE
);

CREATE TABLE modelos (
    id_modelo SERIAL PRIMARY KEY,
    nome_modelo VARCHAR(100) NOT NULL,
    ano_modelo INTEGER NOT NULL,
    id_marca INTEGER NOT NULL,
    CONSTRAINT fk_modelo_marca
        FOREIGN KEY (id_marca)
        REFERENCES marcas(id_marca),
    CONSTRAINT unique_modelo_ano
        UNIQUE (nome_modelo, ano_modelo, id_marca)
);

CREATE TABLE veiculos (
    id_veiculo SERIAL PRIMARY KEY,
    id_usuario INTEGER NOT NULL,
    id_modelo INTEGER NOT NULL,
    CONSTRAINT fk_veiculo_usuario
        FOREIGN KEY (id_usuario)
        REFERENCES usuarios(id_usuario)
        ON DELETE CASCADE,
    CONSTRAINT fk_veiculo_modelo
        FOREIGN KEY (id_modelo)
        REFERENCES modelos(id_modelo)
        ON DELETE RESTRICT
);

SELECT
    v.id_veiculo,
    u.nome_usuario,
    u.tipo_usuario,
    m.nome_modelo,
    m.ano_modelo,
    ma.nome_marca,
    ma.sigla_marca
FROM veiculos v
INNER JOIN usuarios u ON u.id_usuario = v.id_usuario
INNER JOIN modelos m ON m.id_modelo = v.id_modelo
INNER JOIN marcas ma ON ma.id_marca = m.id_marca;
